■ 余白禁止の分割アルゴリズム

function splitRows(n, maxCols) {          // n個を maxCols 以下の行に均等分割（空きセルを作らない）
    const rows = Math.ceil(n / maxCols);
    const base = Math.floor(n / rows), rem = n % rows;
    return Array.from({length: rows}, (_, i) => base + (i < rem ? 1 : 0));
}
// 各行は flex で items を flex:1 にして幅を等分 → 何個でも行がぴったり埋まる
